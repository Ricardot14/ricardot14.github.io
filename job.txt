#!/bin/sh
# Monterey UltraLite diagnostic stage 3 (read-only; macOS Recovery compatible)
# No disk, EFI, SMBIOS, NVRAM, or network modifications.
echo '=== MONTEREY STAGE 3: INSTALLER AND MODEL CHECK ==='
echo '--- Booted OS and date ---'
date
sw_vers 2>&1
echo '--- Native model ---'
sysctl -n hw.model 2>&1
echo '--- Board-ID and OpenCore PlatformInfo ---'
ioreg -rd1 -c IOPlatformExpertDevice 2>&1 | grep -E '"board-id"|"model"|"compatible"' | head -10 || true
echo '--- OpenCore NVRAM hints ---'
nvram -p 2>/dev/null | grep -Ei 'OCLP|opencore|boot-args' | head -12 || true
echo '--- Monterey installer size and signature ---'
for APP in /Volumes/*/'Install macOS Monterey.app'
do
  [ -d "$APP" ] || continue
  echo "FOUND $APP"
  [ -f "$APP/Contents/SharedSupport/SharedSupport.dmg" ] && ls -lh "$APP/Contents/SharedSupport/SharedSupport.dmg"
  /usr/bin/codesign -dv "$APP" 2>&1 | grep -E 'Identifier=|Authority=|TeamIdentifier=|Timestamp=' | head -8 || true
done
echo '--- Target volume: MLU-Monterey ---'
diskutil info '/Volumes/MLU-Monterey' 2>&1 | grep -E 'Volume Name:|Mounted:|Mount Point:|File System Personality:|Disk Size:|Volume Total Space:' || true
echo '--- Relevant errors in install.log ---'
if [ -r /var/log/install.log ]; then
  grep -Ei 'OSISPredicateUpdateProduct|No product found|Code=303|Code=207|Prepare recovery|Copy App Bundle|arrowbuttonPressed|Installation requires|Could not continue|OSInstallerSetup.error|Failed to load trust cache' /var/log/install.log | tail -45 || true
else
  echo 'install.log unavailable'
fi
echo '=== END: READ-ONLY CHECK, NO CHANGES ==='
