#!/bin/sh
# Monterey UltraLite: rebuild installer ONLY on Kingston USB, preserve USB EFI.
# Use from Sierra 10.12.6 on verified iMac10,1. No internal disk changes.
set -eu
export PATH=/usr/bin:/bin:/usr/sbin:/sbin
stop(){ echo "STOP: $*" >&2; exit 1; }
echo '=== MONTEREY KINGSTON USB REBUILD ==='
OS="$(sw_vers -productVersion)" || stop 'Cannot identify OS.'
if [ "$OS" != 10.12.6 ]; then
  echo "You are in macOS $OS. Restart into Macintosh HD (Sierra), then run SAME short link."
  echo 'Nothing changed.'
  exit 2
fi
[ "$(sysctl -n hw.model)" = iMac10,1 ] || stop 'Not the expected 2009 iMac.'
APP='/Applications/Install macOS Monterey.app'
CIM="$APP/Contents/Resources/createinstallmedia"
DMG="$APP/Contents/SharedSupport/SharedSupport.dmg"
RESOURCE="$APP/Contents/Resources/arrowbuttonPressed.tiff"
[ -x "$CIM" ] || stop 'Local Monterey createinstallmedia unavailable. USB untouched.'
[ -f "$DMG" ] || stop 'Source installer DMG missing. USB untouched.'
[ -f "$RESOURCE" ] || stop 'Source installer resource missing. USB untouched.'
/bin/cat "$RESOURCE" >/dev/null || stop 'Source installer file unreadable. USB untouched.'
SIZE="$(stat -f %z "$DMG")"
[ "$SIZE" -gt 10000000000 ] || stop 'Source Monterey DMG unexpectedly small. USB untouched.'
echo 'Verifying original Monterey disk image before changing USB...'
hdiutil verify "$DMG" || stop 'Original Monterey disk image failed verification. USB untouched.'
echo 'SOURCE VERIFIED'
DRIVES="$(diskutil list external physical | awk '/^\/dev\/disk[0-9]+ /{sub("/dev/","",$1); print $1}')"
COUNT="$(printf '%s\n' "$DRIVES" | awk 'NF{n++}END{print n+0}')"
[ "$COUNT" -eq 1 ] || stop 'Connect exactly one external physical USB (Kingston).'
DEV="$DRIVES"
case "$DEV" in disk[1-9]*) ;; *) stop "Unsafe device $DEV";; esac
INFO="$(diskutil info "$DEV")" || stop 'USB info unavailable.'
printf '%s\n' "$INFO" | grep -Eq '^[[:space:]]*Device Location:[[:space:]]*External([[:space:]]|$)' || stop 'Disk not external.'
printf '%s\n' "$INFO" | grep -Eiq '^[[:space:]]*Protocol:[[:space:]]*USB([[:space:]]|$)' || stop 'Disk not USB.'
printf '%s\n' "$INFO" | grep -Eq '^[[:space:]]*Whole:[[:space:]]*Yes([[:space:]]|$)' || stop 'Disk not whole.'
printf '%s\n' "$INFO" | grep -iq DataTraveler || stop 'Not Kingston DataTraveler.'
TMP="$(mktemp -d /tmp/mlu-rebuild.XXXXXX)" || stop 'Cannot create temp folder.'
trap 'rm -rf "$TMP"' EXIT
diskutil info -plist "$DEV" > "$TMP/disk.plist"
CAP="$(/usr/libexec/PlistBuddy -c 'Print :TotalSize' "$TMP/disk.plist")"
case "$CAP" in ''|*[!0-9]*) stop 'Unknown USB size.';; esac
[ "$CAP" -ge 14500000000 ] && [ "$CAP" -le 17000000000 ] || stop "Unexpected USB capacity $CAP."
PART="$DEV"s2
EFI="$DEV"s1
diskutil info "$PART" >/dev/null || stop 'USB installer partition absent.'
diskutil info "$EFI" >/dev/null || stop 'USB EFI partition absent.'
echo "Kingston confirmed: $DEV; volume: $PART; EFI: $EFI"
diskutil mount "$EFI" >/dev/null 2>&1 || true
diskutil info -plist "$EFI" > "$TMP/efi.plist"
EFIMP="$(/usr/libexec/PlistBuddy -c 'Print :MountPoint' "$TMP/efi.plist" 2>/dev/null || true)"
[ -n "$EFIMP" ] && [ -f "$EFIMP/EFI/OC/config.plist" ] && [ -f "$EFIMP/EFI/BOOT/BOOTx64.efi" ] || stop 'Bootable OpenCore EFI not found. No erase.'
BACKUP="$HOME/Documents/Monterey-EFI-Backup-$(date +%Y%m%d-%H%M%S)"
mkdir -p "$BACKUP"
sudo ditto "$EFIMP/EFI" "$BACKUP/EFI" || stop 'Cannot back up EFI.'
cmp "$EFIMP/EFI/OC/config.plist" "$BACKUP/EFI/OC/config.plist" || stop 'EFI backup verification failed.'
echo "OPENCORE EFI BACKED UP: $BACKUP"
diskutil info -plist "$PART" > "$TMP/part.plist"
MP="$(/usr/libexec/PlistBuddy -c 'Print :MountPoint' "$TMP/part.plist" 2>/dev/null || true)"
if [ -z "$MP" ] || [ ! -d "$MP" ]; then
  diskutil mount "$PART" || stop 'Installer partition not mountable.'
  diskutil info -plist "$PART" > "$TMP/part.plist"
  MP="$(/usr/libexec/PlistBuddy -c 'Print :MountPoint' "$TMP/part.plist" 2>/dev/null || true)"
fi
[ -n "$MP" ] && [ -d "$MP" ] || stop 'USB volume mount point missing.'
echo ''
echo "WARNING: This erases only Kingston USB installer partition $PART at $MP."
echo 'Internal Sierra disk and separate MLU-Monterey remain untouched.'
echo 'OpenCore EFI is backed up and must remain on the Kingston USB.'
echo 'Type exactly REBUILD-USB to authorize, or Ctrl+C to cancel:'
read -r ANSWER || stop 'No authorization.'
[ "$ANSWER" = REBUILD-USB ] || stop 'Cancelled. Nothing erased.'
echo 'Apple createinstallmedia rebuilding Kingston installer partition...'
sudo "$CIM" --volume "$MP" --applicationpath "$APP" --nointeraction || stop 'Rebuild failed; Sierra remains untouched.'
diskutil info -plist "$PART" > "$TMP/part-after.plist"
NEWMP="$(/usr/libexec/PlistBuddy -c 'Print :MountPoint' "$TMP/part-after.plist" 2>/dev/null || true)"
[ -n "$NEWMP" ] && [ -d "$NEWMP" ] || stop 'New installer USB not mounted.'
NEWRES="$NEWMP/Install macOS Monterey.app/Contents/Resources/arrowbuttonPressed.tiff"
[ -f "$NEWRES" ] || stop 'New USB missing installer resource.'
/bin/cat "$NEWRES" >/dev/null || stop 'Rebuilt USB file unreadable. Kingston USB hardware is suspect.'
diskutil mount "$EFI" >/dev/null 2>&1 || true
diskutil info -plist "$EFI" > "$TMP/efi-after.plist"
AFTER="$(/usr/libexec/PlistBuddy -c 'Print :MountPoint' "$TMP/efi-after.plist" 2>/dev/null || true)"
[ -n "$AFTER" ] && [ -f "$AFTER/EFI/OC/config.plist" ] && [ -f "$AFTER/EFI/BOOT/BOOTx64.efi" ] || stop "USB installer rebuilt but OpenCore EFI missing. Backup: $BACKUP."
cmp "$BACKUP/EFI/OC/config.plist" "$AFTER/EFI/OC/config.plist" || stop "EFI changed unexpectedly. Backup: $BACKUP."
echo 'Adding offline USB Terminal toolkit (does NOT modify the Apple installer app)...'
cat > "$TMP/mlu.sh" <<'MLU_TOOLBOX'
#!/bin/sh
# Monterey UltraLite: portable offline, read-only diagnostic commands.
# Keep this script OUTSIDE the signed Apple installer app.
MODE=help
[ "$#" -gt 0 ] && MODE="$1"
case "$MODE" in
  help|menu)
    echo '=== MONTEREY USB TOOLBOX (READ ONLY) ==='
    echo 'Run this same command with a final argument:'
    echo '  status  - OS, hardware, disks, volumes'
    echo '  usb     - installer USB resource read test'
    echo '  logs    - latest installation error messages'
    echo '  network - check Apple recovery server'
    echo 'No changes to Sierra, partitions, USB EFI or macOS.'
    ;;
  status)
    echo '=== OS / MODEL ==='
    sw_vers 2>&1
    sysctl hw.model 2>&1
    echo '=== MOUNTED VOLUMES ==='
    ls -1 /Volumes 2>&1
    echo '=== DISK MAP ==='
    diskutil list 2>&1
    ;;
  usb)
    echo '=== INSTALLER RESOURCE READ TEST ==='
    FOUND=0
    for APP in /Volumes/*/'Install macOS Monterey.app'
    do
      SRC="$APP/Contents/Resources/arrowbuttonPressed.tiff"
      [ -f "$SRC" ] || continue
      FOUND=1
      echo "Checking $SRC"
      if /bin/cat "$SRC" >/dev/null; then
        echo 'USB RESOURCE READ: PASS'
      else
        echo 'USB RESOURCE READ: FAIL'
      fi
    done
    [ "$FOUND" -eq 1 ] || echo 'Installer app not visible in mounted volumes.'
    ;;
  logs)
    echo '=== RECENT INSTALLATION ERRORS ==='
    if [ -r /var/log/install.log ]; then
      grep -Ei 'Copy App Bundle|arrowbuttonPressed|Failed to load trust cache|Code=303|Code=207|OSInstallerSetup|OSISPredicate|Failed' /var/log/install.log | tail -35 || true
    else
      echo '/var/log/install.log unavailable.'
    fi
    ;;
  network)
    echo '=== APPLE RECOVERY HTTPS ==='
    curl -ILsS --connect-timeout 8 --max-time 16 -o /dev/null -w 'HTTP %{http_code}\n' https://osrecovery.apple.com/ 2>&1
    ;;
  *)
    echo 'Unknown command. Valid: status, usb, logs, network.'
    exit 2
    ;;
esac
MLU_TOOLBOX
/bin/sh -n "$TMP/mlu.sh" || stop 'Offline toolbox syntax check failed. Installer USB was created; nothing else modified.'
sudo mkdir -p "$NEWMP/MLU-Tools" || stop 'Could not create offline USB toolbox folder.'
sudo ditto "$TMP/mlu.sh" "$NEWMP/MLU-Tools/mlu.sh" || stop 'Could not install offline USB toolbox.'
sudo chmod 755 "$NEWMP/MLU-Tools/mlu.sh" || stop 'Could not make offline USB toolbox readable/executable.'
cmp "$TMP/mlu.sh" "$NEWMP/MLU-Tools/mlu.sh" || stop 'Offline USB toolbox copy failed verification.'
echo 'OFFLINE USB TOOLBOX VERIFIED: MLU-Tools/mlu.sh'
echo "In Monterey Recovery Terminal, run: sh /Volumes/*/MLU-Tools/mlu.sh"
echo 'Optional: append status, usb, logs, or network.'
echo '=== SUCCESS: REBUILT USB INSTALLER + PRESERVED OPENCORE ==='
echo 'Restart holding Option. Select Kingston EFI Boot, then Install macOS Monterey.'
echo 'Use MLU-Monterey as destination. NEVER erase Macintosh HD.'
