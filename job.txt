#!/bin/sh
# Monterey UltraLite: decisive USB media integrity check (v4).
# Does not erase, partition, modify EFI, change SMBIOS, or start an installation.
set -u
echo '=== MONTEREY STAGE 4: USB INTEGRITY ==='
echo 'This is a read-only integrity check of the installer, except a small disposable /tmp test file.'
echo '--- Mounted installers ---'
SRC=''
DMG=''
for APP in /Volumes/*/'Install macOS Monterey.app'
do
  if [ -f "$APP/Contents/Resources/arrowbuttonPressed.tiff" ]; then
    SRC="$APP/Contents/Resources/arrowbuttonPressed.tiff"
    DMG="$APP/Contents/SharedSupport/SharedSupport.dmg"
    echo "Installer: $APP"
    break
  fi
done
if [ -z "$SRC" ]; then
  echo 'STOP: USB Monterey installer resource not mounted. Boot Monterey USB Recovery and retry.'
  exit 2
fi
echo '--- USB mount and image files ---'
df -h "$(dirname "$SRC")" 2>&1 || true
ls -l "$SRC" "$DMG" 2>&1 || true
echo '--- Reproduce the exact resource-copy error safely ---'
TMP="$(mktemp -d /tmp/mlu-tiff.XXXXXX 2>/dev/null)" || { echo 'STOP: Could not create temporary scratch directory.'; exit 3; }
trap 'rm -rf "$TMP"' EXIT HUP INT TERM
if cp -p "$SRC" "$TMP/arrowbuttonPressed.tiff" 2>"$TMP/copyerr"; then
  echo 'PASS: Resource copied with metadata.'
else
  echo 'FAIL: Copy with metadata failed:'
  cat "$TMP/copyerr"
  if cp "$SRC" "$TMP/plain.tiff" 2>"$TMP/plainerr"; then
    echo 'PASS: Plain copy succeeded; source is readable; metadata copy is suspect.'
  else
    echo 'FAIL: Plain copy also failed:'
    cat "$TMP/plainerr"
    echo 'USB SOURCE READ FAILURE. Do not attempt installation using this copy.'
    exit 4
  fi
fi
echo '--- Integrity of Monterey SharedSupport.dmg ---'
if [ ! -f "$DMG" ]; then
  echo 'FAIL: SharedSupport.dmg is missing. Recreate the USB installer.'
  exit 5
fi
echo 'Verifying entire installer image. On USB 2.0 this can be slow; leave the USB connected.'
if hdiutil verify "$DMG"; then
  echo 'RESULT: INSTALLER DMG VERIFIED. Do not rebuild the USB for an assumed corrupt image.'
else
  echo 'RESULT: INSTALLER DMG FAILED VERIFICATION. Recreate the USB installer from an independently verified source.'
  exit 6
fi
echo '=== DONE: Sierra, Monterey target, and USB EFI not modified ==='
