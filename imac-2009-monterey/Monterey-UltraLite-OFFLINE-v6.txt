# experimental no-USB workflow, NOT an unattended or guaranteed installation.
# Important: the original Sierra and shared EFI are on the only physical HDD.
set -euo pipefail
umask 022

BASE="/Users/Shared/Monterey-UltraLite-Offline-v6"
MASTER="/Users/Shared/Monterey-UltraLite-v4/Monterey-UltraLite-Master.command"
GPT_TOOL="/Users/Shared/Monterey-UltraLite-Partition-v5.txt"
MONTEREY_APP="/Applications/Install macOS Monterey.app"
INSTALL_MEDIA="$MONTEREY_APP/Contents/Resources/createinstallmedia"
SIERRA_SECTORS=1367187496
RECOVERY_SECTORS=1269536
INST_START=1368866672
INST_SECTORS=67108864
MON_START=1435975536
MON_SECTORS=517285888
INST_BYTES=$((INST_SECTORS*512))
MON_BYTES=$((MON_SECTORS*512))

say() { printf '%s\n' "$*"; }
stop() { say "STOP: $*" >&2; exit 1; }
need() { [ -x "$1" ] || stop "Missing executable: $1"; }
version() { /usr/bin/sw_vers -productVersion; }
model() { /usr/sbin/sysctl -n hw.model; }
check_model() { [ "$(model)" = iMac10,1 ] || stop "Expected iMac10,1; got $(model)"; }
check_sierra() {
  check_model
  case "$(version)" in 10.12.*) ;; *) stop "This phase requires Sierra 10.12.x; got $(version)";; esac
  [ "$(/usr/sbin/diskutil info / | /usr/bin/sed -n 's/^[[:space:]]*Device Identifier:[[:space:]]*//p' | /usr/bin/head -1)" = disk0s2 ] || stop 'Current boot volume is not Sierra disk0s2.'
}
check_monterey() {
  check_model
  case "$(version)" in 12.*) ;; *) stop "First-boot phase requires Monterey 12.x; got $(version)";; esac
  /usr/sbin/diskutil info / | /usr/bin/grep -qi APFS || stop 'Current boot volume is not APFS.'
  [ "$(id -u)" -ne 0 ] || stop 'Run firstboot as the normal user, not with sudo.'
}
confirm() {
  say "Enter EXACTLY: $1"
  local response
  IFS= read -r response || stop 'No confirmation.'
  [ "$response" = "$1" ] || stop 'Confirmation not supplied.'
}

# Only accept the GPT geometry from the known 2026-11-01 full disk report.
gpt_text() { /usr/bin/sudo /usr/sbin/gpt -r show disk0; }
assert_gpt_entry() {
  printf '%s\n' "$GPT" | /usr/bin/awk -v b="$1" -v sz="$2" -v i="$3" -v typ="$4" \
    '$1==b && $2==sz && $3==i && $4=="GPT" && $5=="part" && $7==typ {ok=1} END{exit !ok}' || stop "GPT partition #$3 differs from expected geometry."
}
assert_gpt() {
  GPT="$(gpt_text)" || stop 'Unable to read GPT.'
  assert_gpt_entry 40 409600 1 C12A7328-F81F-11D2-BA4B-00A0C93EC93B
  assert_gpt_entry 409640 "$SIERRA_SECTORS" 2 48465300-0000-11AA-AA11-00306543ECAC
  assert_gpt_entry 1367597136 "$RECOVERY_SECTORS" 3 426F6F74-0000-11AA-AA11-00306543ECAC
}
assert_new_entries() {
  assert_gpt
  assert_gpt_entry "$INST_START" "$INST_SECTORS" 4 48465300-0000-11AA-AA11-00306543ECAC
  assert_gpt_entry "$MON_START" "$MON_SECTORS" 5 48465300-0000-11AA-AA11-00306543ECAC
  local count
  count="$(printf '%s\n' "$GPT" | /usr/bin/awk '$4=="GPT" && $5=="part"{n++} END{print n+0}')"
  [ "$count" = 5 ] || stop "Expected five GPT partitions, found $count."
}

check() {
  check_model
  say "=== OFFLINE v6 / $(version) ==="
  /usr/sbin/diskutil list disk0
  say '=== GPT (read only, may request admin password) ==='
  gpt_text || true
  say '=== Installers and staged tools ==='
  /bin/ls -ld "$MONTEREY_APP" "$MASTER" "$GPT_TOOL" "$BASE" 2>&1 || true
  say '=== Recovery/backups ==='
  /usr/bin/tmutil latestbackup 2>&1 || true
}

prepare() {
  check_sierra
  [ "$(id -u)" -ne 0 ] || stop 'Run prepare WITHOUT sudo.'
  [ -f "$GPT_TOOL" ] || stop 'Partition v5 not staged; run: sudo bash ~/Downloads/Monterey-UltraLite-Partition-v5.txt stage'
  [ -f "$MASTER" ] || stop 'Master v4.1 not staged; run its stage command first.'
  [ -x "$INSTALL_MEDIA" ] || stop 'Monterey installer missing or incomplete.'
  assert_gpt
  /bin/mkdir -p "$BASE"
  /bin/cp "$0" "$BASE/Monterey-UltraLite-OFFLINE-v6.txt"
  /bin/chmod 644 "$BASE/Monterey-UltraLite-OFFLINE-v6.txt"
  if [ -s "$HOME/Monterey-UltraLite/OpenCore-Patcher-2.5.1.pkg" ]; then
    /bin/cp "$HOME/Monterey-UltraLite/OpenCore-Patcher-2.5.1.pkg" "$BASE/OpenCore-Patcher.pkg"
    say 'OCLP installer package copied for use from Monterey.'
  else
    say 'WARNING: Local OCLP package not found. Monterey post-install root patches may require a new download.'
  fi
  /usr/bin/sed -n "/^: <<'MLU_EMBEDDED_GUIDE'$/,/^MLU_EMBEDDED_GUIDE$/p" "$0" | /usr/bin/sed '1d;$d' > "$BASE/OFFLINE-GUIDE.txt"
  [ -s "$BASE/OFFLINE-GUIDE.txt" ] || stop 'Embedded guide extraction failed; no reboot yet.'
  /bin/cp "$BASE/OFFLINE-GUIDE.txt" "$HOME/Desktop/Monterey-UltraLite-OFFLINE-GUIDE.txt"
  say 'Complete offline guide installed on Sierra Desktop.'
  say "OFFLINE SCRIPT SAVED: $BASE/Monterey-UltraLite-OFFLINE-v6.txt"
  say 'Now use the OFFLINE GUIDE. This step changed NO partitions or EFI.'
}

recovery() {
  # Run only from Recovery Terminal, from a copy in /tmp.
  check_model
  [ "$(id -u)" -eq 0 ] || stop 'Recovery provisioning requires a root shell.'
  current="$(/usr/sbin/diskutil info / 2>/dev/null | /usr/bin/sed -n 's/^[[:space:]]*Device Identifier:[[:space:]]*//p' | /usr/bin/head -1)"
  [ "$current" != disk0s2 ] || stop 'You are booted into Sierra; do not write GPT here.'
  [ -f /tmp/mlu-v5.txt ] || stop 'Missing staged v5 in /tmp. Consult offline guide.'
  say 'Delegating partition operation to validated-geometry v5 script.'
  /bin/bash /tmp/mlu-v5.txt provision
  say 'GPT attempt finished. Read the exact v5 output and reboot to Sierra only if both entries verified.'
}

# Finds an ONLY-on-disk0 HFS+ partition of the exact known byte size.
# Disk Utility sometimes presents physical slice numbers in a different order
# from GPT indexes; we NEVER derive slice IDs from GPT indices.
identify_partition() {
  local expected="$1" candidate bytes matches="" count=0
  for candidate in $(/usr/sbin/diskutil list disk0 | /usr/bin/awk '/Apple_HFS/ {print $NF}' | /usr/bin/grep '^disk0s[0-9][0-9]*$'); do
    case "$candidate" in disk0s1|disk0s2|disk0s4) continue;; esac
    bytes="$(/usr/sbin/diskutil info "$candidate" | /usr/bin/awk -F '[()]' '/Disk Size:/ {split($2,p," ");print p[1];exit}')"
    if [ "$bytes" = "$expected" ]; then
      matches="$candidate"
      count=$((count+1))
    fi
  done
  [ "$count" -eq 1 ] || stop "Could not uniquely identify new disk0 slice of $expected bytes (found $count). STOP."
  printf '%s\n' "$matches"
}

# Verify a named volume by exact size, never by name alone.
verify_named() {
  local name="$1" expected="$2" info id bytes
  info="$(/usr/sbin/diskutil info "/Volumes/$name")" || stop "$name is not mounted."
  id="$(printf '%s\n' "$info" | /usr/bin/sed -n 's/^[[:space:]]*Device Identifier:[[:space:]]*//p' | /usr/bin/head -1)"
  case "$id" in disk0s1|disk0s2|disk0s4|disk1*|'') stop "Unsafe target identifier $id for $name";; esac
  case "$id" in disk0s*) ;; *) stop "$name must be on disk0";; esac
  bytes="$(printf '%s\n' "$info" | /usr/bin/awk -F '[()]' '/Disk Size:/ {split($2,p," ");print p[1];exit}')"
  [ "$bytes" = "$expected" ] || stop "$name unexpected size ($bytes vs $expected)."
  printf '%s\n' "$id"
}

efi_backup() {
  local mountpoint
  /usr/bin/sudo /usr/sbin/diskutil mount disk0s1 >/dev/null || true
  mountpoint="$(/usr/sbin/diskutil info disk0s1 | /usr/bin/sed -n 's/^[[:space:]]*Mount Point:[[:space:]]*//p' | /usr/bin/head -1)"
  [ -n "$mountpoint" ] && [ -d "$mountpoint" ] || stop 'Could not mount EFI; do not install OpenCore.'
  /bin/mkdir -p "$BASE/EFI-BACKUP-BEFORE-OCLP"
  /usr/bin/sudo /usr/bin/ditto "$mountpoint" "$BASE/EFI-BACKUP-BEFORE-OCLP" || stop 'EFI backup failed.'
  say "EFI copied to $BASE/EFI-BACKUP-BEFORE-OCLP (SAME DISK; not a disaster backup)"
}

# On Sierra after Recovery GPT operations. This deliberately does not run OCLP
# or touch the shared EFI except mounting it READ-ONLY for backup copies.
finish() {
  check_sierra
  [ "$(id -u)" -ne 0 ] || stop 'Run finish without sudo.'
  [ -x "$INSTALL_MEDIA" ] || stop 'Monterey createinstallmedia not found.'
  assert_new_entries
  [ -f "$BASE/Monterey-UltraLite-OFFLINE-v6.txt" ] || stop 'Run prepare first.'
  # A previous invocation might already have created the Apple installer.
  # Never overwrite it on a re-run.
  if /usr/sbin/diskutil info '/Volumes/Install macOS Monterey' >/dev/null 2>&1; then
    verify_named 'Install macOS Monterey' "$INST_BYTES" >/dev/null
    /usr/sbin/diskutil info '/Volumes/MLU-Monterey' >/dev/null 2>&1 || stop 'Installer exists but Monterey destination is missing.'
    verify_named 'MLU-Monterey' "$MON_BYTES" >/dev/null
    say 'Apple installer and Monterey destination already exist. No formatting or writing.'
    efi_backup
    /usr/bin/open -a OpenCore-Patcher || true
    return 0
  fi
  say 'Both new GPT entries verified. Formatting only the two new partitions.'
  confirm 'FORMAT ONLY THE TWO NEW PARTITIONS'
  local installer_id monterey_id
  installer_id="$(identify_partition "$INST_BYTES")"
  monterey_id="$(identify_partition "$MON_BYTES")"
  [ "$installer_id" != "$monterey_id" ] || stop 'Partition identifiers collide.'
  say "Exact-size slice IDs: installer=$installer_id, Monterey=$monterey_id"
  # Don't erase a previously initialized volume if finish has been rerun.
  if /usr/sbin/diskutil info /Volumes/MLU-Installer >/dev/null 2>&1; then
    [ "$(verify_named MLU-Installer "$INST_BYTES")" = "$installer_id" ] || stop 'Installer name/device mismatch.'
    say 'Installer volume already formatted; leaving intact.'
  else
    /usr/bin/sudo /usr/sbin/diskutil eraseVolume JHFS+ MLU-Installer "$installer_id" || stop 'Installer formatting failed. STOP.'
  fi
  if /usr/sbin/diskutil info /Volumes/MLU-Monterey >/dev/null 2>&1; then
    [ "$(verify_named MLU-Monterey "$MON_BYTES")" = "$monterey_id" ] || stop 'Monterey name/device mismatch.'
    say 'Monterey volume already formatted; leaving intact.'
  else
    /usr/bin/sudo /usr/sbin/diskutil eraseVolume JHFS+ MLU-Monterey "$monterey_id" || stop 'Monterey formatting failed. STOP.'
  fi
  [ "$(verify_named MLU-Installer "$INST_BYTES")" = "$installer_id" ] || stop 'Installer identity changed.'
  [ "$(verify_named MLU-Monterey "$MON_BYTES")" = "$monterey_id" ] || stop 'Monterey identity changed.'
  efi_backup
  if [ -e '/Volumes/Install macOS Monterey' ]; then
    say 'An installer volume already exists. Not rewriting it.'
  else
    say 'Apple createinstallmedia will ERASE ONLY MLU-Installer, verified by physical size and slice.'
    confirm 'CREATE MONTEREY INSTALLER'
    /usr/bin/sudo "$INSTALL_MEDIA" --volume /Volumes/MLU-Installer --applicationpath "$MONTEREY_APP" --nointeraction || stop 'createinstallmedia failed. Do not erase Sierra or rerun partitioning.'
  fi
  say 'NEXT: OPENCORE GUI is required to build an EFI boot configuration for iMac10,1.'
  say 'Consult the offline guide BEFORE rebooting. OpenCore will modify the shared internal EFI.'
  /usr/bin/open -a OpenCore-Patcher || true
}


install_patcher() {
  check_monterey
  local mounted package
  mounted="$(/usr/sbin/diskutil info disk0s2 | /usr/bin/sed -n 's/^[[:space:]]*Mount Point:[[:space:]]*//p' | /usr/bin/head -1)"
  if [ -z "$mounted" ] || [ "$mounted" = 'Not mounted' ]; then
    /usr/sbin/diskutil mount disk0s2 || stop 'Could not mount Sierra.'
    mounted="$(/usr/sbin/diskutil info disk0s2 | /usr/bin/sed -n 's/^[[:space:]]*Mount Point:[[:space:]]*//p' | /usr/bin/head -1)"
  fi
  package="$mounted/Users/Shared/Monterey-UltraLite-Offline-v6/OpenCore-Patcher.pkg"
  [ -s "$package" ] || stop 'Staged OCLP package not found. Locate the original OCLP PKG on Sierra.'
  /usr/sbin/pkgutil --check-signature "$package" || stop 'Package signature check failed.'
  say 'This installs the downloaded OCLP package into CURRENT MONTEREY only, not Sierra.'
  confirm 'INSTALL OPENCORE PATCHER IN MONTEREY'
  /usr/bin/sudo /usr/sbin/installer -pkg "$package" -target / || stop 'OCLP package installation failed.'
  /usr/bin/open -a OpenCore-Patcher || true
  say 'In OCLP: Post-Install Root Patch -> Start Root Patching -> Reboot.'
}

firstboot() {
  check_monterey
  # Sierra HFS+ must be mounted from Monterey to reach existing stage files.
  local mounted
  mounted="$(/usr/sbin/diskutil info disk0s2 | /usr/bin/sed -n 's/^[[:space:]]*Mount Point:[[:space:]]*//p' | /usr/bin/head -1)"
  if [ -z "$mounted" ] || [ "$mounted" = 'Not mounted' ]; then
    /usr/sbin/diskutil mount disk0s2 || stop 'Could not mount Sierra volume.'
    mounted="$(/usr/sbin/diskutil info disk0s2 | /usr/bin/sed -n 's/^[[:space:]]*Mount Point:[[:space:]]*//p' | /usr/bin/head -1)"
  fi
  [ -f "$mounted/Users/Shared/Monterey-UltraLite-v4/Monterey-UltraLite-Master.command" ] || stop 'Staged Monterey master not found on Sierra.'
  /bin/bash "$mounted/Users/Shared/Monterey-UltraLite-v4/Monterey-UltraLite-Master.command" firstboot
  say 'First-boot preference profile and desktop controls installed.'
  say 'This does NOT enable unverified Target Display Mode, AirPlay 2 receiver, Handoff or Universal Control.'
}
