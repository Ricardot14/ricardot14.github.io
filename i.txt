#!/bin/sh
# Monterey UltraLite stable launcher. Always pull current job with cache buster.
# This launcher never changes disks or EFI.
set -eu
echo 'MONTEREY LAUNCHER v1: fetching newest diagnostic'
URL="https://raw.githubusercontent.com/Ricardot14/ricardot14.github.io/master/job.txt?v=$(date +%s)"
OUT="/tmp/mlu-job.$$"
trap 'rm -f "$OUT"' EXIT HUP INT TERM
curl -fLsS --connect-timeout 10 --max-time 90 "$URL" -o "$OUT" || { echo 'Download failed; nothing executed.'; exit 1; }
head -n 1 "$OUT" | grep -q '^#!/bin/sh' || { echo 'Invalid script content; nothing executed.'; exit 1; }
echo 'Latest job downloaded. Executing:'
/bin/sh "$OUT"
